program TestLogic;
var
  x : integer;
  y : integer;
  flag : boolean;
begin
  x := 5;
  y := 10;
  flag := (x < y) and (y > 0);
  if flag then
    writeln(1)
  else
    writeln(0);
  flag := not flag;
  if flag then
    writeln(1)
  else
    writeln(0);
  flag := (x > y) or (x = 5);
  if flag then
    writeln(1)
  else
    writeln(0)
end.
